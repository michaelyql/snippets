
# Read Write Lock

```cpp
class ReaderWriterLock
{
public:
    unsigned ReaderLock()
    {
        std::unique_lock<std::mutex> lk(mtx);
        readers_ready.wait(lk, [this]{
            return !pendingWriters && !writers;
        });
        return readers++;
    }

    unsigned ReaderUnlock() 
    { 
        std::unique_lock<std::mutex> lk(mtx);
        if (!--readers && pendingWriters) {
            writers_ready.notify_one();  
        } 
        return readers;
    }

    void WriterLock() 
    { 
        std::unique_lock<std::mutex> lk(mtx);
        ++pendingWriters;
        writers_ready.wait(lk, [this]{
            return !readers && !writers;
        });
        --pendingWriters; 
        writers++;
    }

    void WriterUnlock()
    { 
        std::unique_lock<std::mutex> lk(mtx);
        writers--;
        if (pendingWriters) writers_ready.notify_one();
        else readers_ready.notify_one();
    }

private:
    std::mutex mtx;
    std::condition_variable readers_ready, writers_ready;
    size_t readers {0}, writers {0}, pendingWriters {0};
};
```

- Think of the mutex as a "block"
    - Successfully acquiring the lock on the mutex means that operation succeeded (be it reader lock/unlock, writer lock/unlock)
    - Unlock the mutex so that other threads can enter other functions and acquire the lock
- `pendingWriters` is an optimization to favour writes
