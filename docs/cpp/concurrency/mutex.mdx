# Mutex

Possible implementation: 

```cpp
#include <atomic>

class Mutex {
    std::atomic_flag m_{};
public:
    void lock() noexcept {
        while (m_.test_and_set(std::memory_order_acq_rel)) {
            m_.wait(true, std::memory_order_acquire);
        }
    }
    bool try_lock() noexcept {
        return !m_.test_and_set(std::memory_order_acquire);
    }
    void unlock() noexcept {
        m_.clear(std::memory_order_release);
        m_.notify_one();
    }
};
```

Alternative implementation
```cpp
class Mutex {
    std::atomic<int> m_{0};
public:
    void lock() noexcept {
        while (m_.load(std::memory_order_relaxed) || m_.exchange(1, std::memory_order_acquire)) {}
    }
    bool try_lock() noexcept {
        return !m_.exchange(1, std::memory_order_acquire);
    }
    void unlock() noexcept {
        m_.store(0, std::memory_order_release);
    }
};
```
