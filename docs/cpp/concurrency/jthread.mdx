# jthread

```cpp 
#include <thread>
#include <memory>
#include <functional>
#include <concepts>

class stop_source;

class stop_token {
public:
    stop_token() = default;

    bool stop_requested() const {
        return token_ptr_->load(std::memory_order_acquire);
    }

private:
    bool cancel() {
        return !token_ptr_->exchange(true, std::memory_order_acq_rel);
    }

    friend class stop_source;

    std::shared_ptr<std::atomic<bool>> token_ptr_{
        std::make_shared<std::atomic<bool>>(false)
    };
};

class stop_source {
public:
    stop_source() = default;

    bool request_stop() {
        return token_.cancel();
    }

    bool stop_requested() const {
        return token_.stop_requested();
    }

    stop_token get_token() {
        return token_;
    }

private:
    stop_token token_;
};

template <typename F, typename... Args>
concept AcceptsStopToken = std::invocable<F, stop_token, Args...>;

class jthread {
public:
    ~jthread() {
        request_stop();
        if (joinable()) {
            join();
        }
    }

    jthread(const jthread&) = delete;
    jthread& operator=(const jthread&) = delete;
    jthread(jthread&&) = delete;
    jthread& operator=(jthread&&) = delete;

    template <class F, class... Args>
    explicit jthread(F&& f, Args&&... args) {
        auto token = source_.get_token();

        if constexpr (AcceptsStopToken<F, Args...>) {
            thread_ = std::thread(
                [token, function = std::forward<F>(f)](Args&&... arguments) mutable {
                    std::invoke(function, token, std::forward<Args>(arguments)...);
                },
                std::forward<Args>(args)...
            );
        } else {
            thread_ = std::thread(
                [function = std::forward<F>(f)](Args&&... arguments) mutable {
                    std::invoke(function, std::forward<Args>(arguments)...);
                },
                std::forward<Args>(args)...
            );
        }
    }

    bool joinable() const {
        return thread_.joinable();
    }

    void join() {
        thread_.join();
    }

    void detach() {
        if (!joinable()) {
            return;
        }
        thread_.detach();
    }

    void request_stop() {
        if (source_.stop_requested()) {
            return;
        }
        source_.request_stop();
    }

private:
    std::thread thread_;
    stop_source source_;
};
```
