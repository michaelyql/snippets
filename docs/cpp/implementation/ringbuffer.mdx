# Ring Buffer 

Non thread-safe ring buffer. Note that the implementation does not clean up elements on `reset()` nor in the destructor

```cpp 
template <typename T>
class CircularBuffer {
public:
    CircularBuffer(): capacity_(MAX_CAPACITY) {
        buffer_ = static_cast<T*>(malloc(sizeof(T) * MAX_CAPACITY));
    }
    explicit CircularBuffer(std::size_t capacity): capacity_(capacity) {
        if (capacity > std::numeric_limits<std::size_t>::max() / sizeof(T)) {
            throw std::bad_alloc{};
        }
        buffer_ = static_cast<T*>(malloc(sizeof(T) * capacity));
    }

    std::shared_ptr<T> read() {
        if (empty()) return std::shared_ptr<T>{};
        auto ptr = std::make_shared<T>(*(buffer_ + tail_));
        tail_ = (tail_ + 1) % capacity_;
        size_ -= 1;
        return ptr;
    }
    bool try_read(T& item) {
        if (empty()) return false;
        item = std::move(*(buffer_ + tail_));
        tail_ = (tail_ + 1) % capacity_;
        size_ -= 1;
        return true;
    }
    void write(const T& item) {
        new (buffer_ + head_) T{item};
        head_ = (head_ + 1) % capacity_;
        size_++;
        if (size_ > capacity_) {
            size_ = capacity_;
            tail_ = (tail_ + 1) % capacity_;
        }
    }
    void reset() {
        size_ = head_ = tail_ = 0;
    }

    [[nodiscard]] bool full()  const {
        return size_ == capacity_;
    }
    [[nodiscard]] bool empty() const {
        return size_ == 0;
    }
    [[nodiscard]] std::size_t size() const {
        return size_;
    }
    [[nodiscard]] std::size_t capacity() const {
        return capacity_;
    }

private:
    T* buffer_;
    const size_t capacity_;
    size_t size_ {0};
    size_t head_ {0}, tail_ {0};

    static constexpr std::size_t MAX_CAPACITY = 
        std::numeric_limits<std::size_t>::max() / sizeof(T);
};
```

Alternative implementation:
```cpp 
template <typename T>
class CircularBuffer {
public:
    CircularBuffer() : CircularBuffer(0) {}
    explicit CircularBuffer(std::size_t capacity)
        : capacity_{ capacity }, ptr_{ std::make_unique<T[]>(capacity) } {}

    std::shared_ptr<T> read() {
        if (size_ == 0) return nullptr;
        auto item = std::make_shared<T>(ptr_[read_idx_]);
        increment_read();
        return item;
    }

    bool try_read(T& item) {
        if (size_ == 0) return false;
        item = ptr_[read_idx_];
        increment_read();
        return true;
    }

    void write(const T& item) {
        ptr_[write_idx_] = item;
        increment_write();
    }

    void reset() {
        size_ = 0;
        write_idx_ = read_idx_;
    }

    [[nodiscard]] bool full()  const { return size_ == capacity_; }
    [[nodiscard]] bool empty() const { return size_ == 0; }
    [[nodiscard]] std::size_t size() const { return size_; }
    [[nodiscard]] std::size_t capacity() const { return capacity_; }

private:
    void increment_read() {
        read_idx_ = (read_idx_ + 1) % capacity_;
        size_--;
    }

    void increment_write() {
        auto next_write = (write_idx_ + 1) % capacity_;
        if (next_write == read_idx_ && size_ == capacity_ - 1)
            read_idx_ = (read_idx_ + 1) % capacity_;
        write_idx_ = next_write;
        size_ = std::min(size_ + 1, capacity_);
    }

    std::size_t capacity_{};
    std::size_t size_{};
    std::size_t read_idx_{};
    std::size_t write_idx_{};
    std::unique_ptr<T[]> ptr_{ nullptr };
};
```
