# Hashmap

## Open Addressing + Linear Probing 

```cpp
template <typename Key, typename Value>
class OpenAddressingHashMap {
private:
    struct Bucket {
        Key key{};
        Value value{};
        bool occupied{ false };
        bool deleted{ false };
    };

    static constexpr std::size_t npos = std::numeric_limits<size_t>::max();

    std::size_t size_;
    std::size_t capacity_;
    std::vector<Bucket> buckets_;

    std::size_t hash(const Key& key) const {
        return std::hash<Key>{}(key) % capacity_;
    }

    std::size_t find_index(const Key& key) const {
        std::size_t idx = hash(key);

        while (buckets_[idx].occupied || buckets_[idx].deleted) {
            if (buckets_[idx].occupied && buckets_[idx].key == key) {
                return idx;
            }
            idx = (idx + 1) % capacity_;
        }

        return npos;
    }

    bool should_resize() const {
        return size_ * 2 >= capacity_;
    }

    void resize() {
        const auto max_capacity = std::numeric_limits<std::size_t>::max() / 2;

        if (capacity_ >= max_capacity) {
            throw std::logic_error("Capacity limit reached");
        }

        std::size_t new_capacity = capacity_ * 2;
        std::vector<Bucket> new_buckets(new_capacity);

        for (auto& bucket : buckets_) {
            if (bucket.occupied) {
                // Must rehash against new_capacity, not old capacity
                std::size_t idx = std::hash<Key>{}(bucket.key) % new_capacity;

                while (new_buckets[idx].occupied) {
                    idx = (idx + 1) % new_capacity;
                }

                new_buckets[idx].key = std::move(bucket.key);
                new_buckets[idx].value = std::move(bucket.value);
                new_buckets[idx].occupied = true;
            }
        }

        buckets_ = std::move(new_buckets);
        capacity_ = new_capacity;
    }

public:
    OpenAddressingHashMap() : OpenAddressingHashMap(16) {}

    explicit OpenAddressingHashMap(std::size_t initial_capacity)
        : size_{0}, capacity_{initial_capacity}, buckets_(initial_capacity) {}

    void insert(const Key& key, const Value& value) {
        if (should_resize()) {
            resize();
        }

        std::size_t idx = hash(key);
        std::size_t first_tombstone = npos;

        // Probe past both occupied and tombstone slots to find key or an empty slot
        while (buckets_[idx].occupied || buckets_[idx].deleted) {
            if (buckets_[idx].occupied && buckets_[idx].key == key) {
                buckets_[idx].value = value;
                return;
            }
            if (buckets_[idx].deleted && first_tombstone == npos) {
                first_tombstone = idx;
            }
            idx = (idx + 1) % capacity_;
        }

        // Reuse the first tombstone we passed instead of the empty slot we landed on
        if (first_tombstone != npos) {
            idx = first_tombstone;
        }

        buckets_[idx].key = key;
        buckets_[idx].value = value;
        buckets_[idx].occupied = true;
        buckets_[idx].deleted = false;
        ++size_;
    }

    bool erase(const Key& key) {
        std::size_t idx = find_index(key);
        if (idx == npos) {
            return false;
        }
        buckets_[idx].occupied = false;
        buckets_[idx].deleted = true;
        --size_;
        return true;
    }

    Value& get(const Key& key) {
        std::size_t idx = find_index(key);
        if (idx == npos) {
            throw std::out_of_range("Key not found");
        }
        return buckets_[idx].value;
    }

    const Value& get(const Key& key) const {
        std::size_t idx = find_index(key);
        if (idx == npos) {
            throw std::out_of_range("Key not found");
        }
        return buckets_[idx].value;
    }

    bool contains(const Key& key) const {
        return find_index(key) != npos;
    }

    std::size_t get_size() const {
        return size_;
    }

    std::size_t get_capacity() const {
        return capacity_;
    }
};
```
